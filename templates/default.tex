% Cross-referencing with an external document (e.g. supplementary material).
% Uncomment \myexternaldocument; latexmkrc builds the external .aux.
\makeatletter
\newcommand*{\addFileDependency}[1]{
\typeout{(#1)}
\@addtofilelist{#1}
\IfFileExists{#1}{}{\typeout{No file #1.}}
}\makeatother
\newcommand*{\myexternaldocument}[2]{
\externaldocument[#1]{#2}
\addFileDependency{#2.tex}
\addFileDependency{#2.aux}
}
